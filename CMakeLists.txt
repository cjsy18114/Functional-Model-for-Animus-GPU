cmake_minimum_required(VERSION 3.16)
project(animus_functional_model VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/softfloat/sources.txt" SOFTFLOAT_FILES)
list(TRANSFORM SOFTFLOAT_FILES PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/third_party/softfloat/")
add_library(animus_softfloat STATIC ${SOFTFLOAT_FILES})
target_include_directories(animus_softfloat PUBLIC third_party/softfloat/source/include
    PRIVATE third_party/softfloat third_party/softfloat/source/8086-SSE)
target_compile_definitions(animus_softfloat PUBLIC SOFTFLOAT_FAST_INT64)
add_library(animus_model src/common.cpp src/memory.cpp src/loader.cpp src/decode.cpp src/engine.cpp src/trace.cpp src/exec_integer.cpp src/exec_memory.cpp src/exec_simt.cpp src/scheduler.cpp src/exec_fp.cpp src/fp_backend.cpp src/exec_csr.cpp)
target_include_directories(animus_model PUBLIC include)
target_link_libraries(animus_model PRIVATE animus_softfloat)
if(MSVC)
  target_compile_options(animus_model PRIVATE /W4 /utf-8)
else()
  target_compile_options(animus_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(animus_model_cli src/main.cpp)
target_link_libraries(animus_model_cli PRIVATE animus_model)
enable_testing()
add_executable(animus_model_tests tests/model_tests.cpp)
target_link_libraries(animus_model_tests PRIVATE animus_model)
add_test(NAME stage_a COMMAND animus_model_tests)

add_executable(animus_model_tests_b tests/stage_b_tests.cpp)
target_link_libraries(animus_model_tests_b PRIVATE animus_model)
add_test(NAME stage_b COMMAND animus_model_tests_b)

add_executable(animus_model_tests_c tests/stage_c_tests.cpp)
target_link_libraries(animus_model_tests_c PRIVATE animus_model)
add_test(NAME stage_c COMMAND animus_model_tests_c)
